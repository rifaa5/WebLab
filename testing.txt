from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait, Select
from selenium.webdriver.support import expected_conditions as EC

driver = webdriver.Chrome()
wait = WebDriverWait(driver, 10)

try:
    
    driver.get("http://127.0.0.1:3000")

   
    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//h2[contains(text(),'Student Management')]")
    ))

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//input[@placeholder='Student Name']")
    ))

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//select[@ng-model='student.department']")
    ))

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//input[@placeholder='Year']")
    ))

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//button[@type='submit']")
    ))

    print("Interface Testing: PASS")

   
    name = driver.find_element(
        By.XPATH, "//input[@placeholder='Student Name']"
    )
    name.send_keys("SeleniumTest")

    department = Select(driver.find_element(
        By.XPATH, "//select[@ng-model='student.department']"
    ))
    department.select_by_visible_text("IT")

    year = driver.find_element(
        By.XPATH, "//input[@placeholder='Year']"
    )
    year.send_keys("3")

    driver.find_element(
        By.XPATH, "//button[@type='submit']"
    ).click()

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//td[text()='SeleniumTest']")
    ))

    print("Functional Testing: PASS")

  
    name = driver.find_element(
        By.XPATH, "//input[@placeholder='Student Name']"
    )
    name.clear()

    year.clear()

    driver.find_element(
        By.XPATH, "//button[@type='submit']"
    ).click()

    if name.get_attribute("required") is not None:
        print("Negative Testing: PASS")
    else:
        print("Negative Testing: FAIL")

   
    name.send_keys("SeleniumTest")

    wait.until(EC.visibility_of_element_located(
        (By.XPATH, "//td[text()='SeleniumTest']")
    ))

    print("Usability Testing: PASS")

    print("All Selenium tests completed.")

finally:
    driver.quit()
