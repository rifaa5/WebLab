XML-dtd 

students.xml

<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE students SYSTEM "students.dtd">

<students>
    <student>
        <name>Riya</name>
        <department>IT</department>
        <year>3</year>
    </student>

    <student>
        <name>Arun</name>
        <department>CSE</department>
        <year>2</year>
    </student>
</students>

students.dtd

<!ELEMENT students (student+)>
<!ELEMENT student (name, department, year)>
<!ELEMENT name (#PCDATA)>
<!ELEMENT department (#PCDATA)>
<!ELEMENT year (#PCDATA)>


validate.py

from lxml import etree

parser = etree.XMLParser(dtd_validation=True)

try:
    etree.parse("students.xml", parser)
    print("XML is valid")
except etree.XMLSyntaxError as e:
    print("XML is invalid")
    print(e)



cmd
pip install lxml
python validate.py
